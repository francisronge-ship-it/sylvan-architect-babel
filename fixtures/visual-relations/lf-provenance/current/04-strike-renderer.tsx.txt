      lf.strikeNodes?.forEach((nodeId) => {
        const elements = findVisualElementsForSubtree(nodeId);
        elements.forEach((element) => element.classList.add('babel-lf-copy-label'));
        const rect = getLocalRectForElements(treeGroup, elements);
        if (!rect) return;
        const y = rect.y + rect.height * 0.56;
        const shadow = document.createElementNS('http://www.w3.org/2000/svg', 'line');
        shadow.classList.add('babel-lf-strike-shadow');
        shadow.setAttribute('x1', (rect.x - 14).toFixed(1));
        shadow.setAttribute('x2', (rect.x + rect.width + 14).toFixed(1));
        shadow.setAttribute('y1', y.toFixed(1));
        shadow.setAttribute('y2', y.toFixed(1));
        layer.appendChild(shadow);

        const strike = document.createElementNS('http://www.w3.org/2000/svg', 'line');
        strike.classList.add('babel-lf-strike');
        strike.setAttribute('x1', (rect.x - 14).toFixed(1));
        strike.setAttribute('x2', (rect.x + rect.width + 14).toFixed(1));
        strike.setAttribute('y1', y.toFixed(1));
        strike.setAttribute('y2', y.toFixed(1));
        layer.appendChild(strike);
      });
